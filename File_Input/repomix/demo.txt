 1: package com.ewolff.microservice.catalog.web;

13: public class CatalogController {

16: 	public CatalogController(ItemRepository itemRepository) {
17: 		this.itemRepository = itemRepository;
18: 	}
19: 	@RequestMapping(value = "/{id}.html", method = RequestMethod.GET, produces = MediaType.TEXT_HTML_VALUE)
20: 	public ModelAndView Item(@PathVariable("id") long id) {
21: 		return new ModelAndView("item", "item", itemRepository.findById(id).get());
22: 	}
23: 	@RequestMapping(value = "/list.html", method = RequestMethod.GET)
24: 	public ModelAndView ItemList() {
25: 		return new ModelAndView("itemlist", "items", itemRepository.findAll());
26: 	}
27: 	@RequestMapping(value = "/form.html", method = RequestMethod.GET)
28: 	public ModelAndView add() {
29: 		return new ModelAndView("item", "item", new Item());
30: 	}
31: 	@RequestMapping(value = "/form.html", method = RequestMethod.POST)
32: 	public ModelAndView post(Item Item) {
33: 		Item = itemRepository.save(Item);
34: 		return new ModelAndView("success");
35: 	}
36: 	@RequestMapping(value = "/{id}.html", method = RequestMethod.PUT)
37: 	public ModelAndView put(@PathVariable("id") long id, Item item) {
38: 		item.setId(id);
39: 		itemRepository.save(item);
40: 		return new ModelAndView("success");
41: 	}
42: 	@RequestMapping(value = "/searchForm.html", produces = MediaType.TEXT_HTML_VALUE, method = RequestMethod.GET)
43: 	public ModelAndView searchForm() {
44: 		return new ModelAndView("searchForm");
45: 	}
46: 	@RequestMapping(value = "/searchByName.html", produces = MediaType.TEXT_HTML_VALUE, method = RequestMethod.GET)
47: 	public ModelAndView search(@RequestParam("query") String query) {
48: 		return new ModelAndView("itemlist", "items", itemRepository.findByNameContaining(query));
49: 	}
50: 	@RequestMapping(value = "/{id}.html", method = RequestMethod.DELETE)
51: 	public ModelAndView delete(@PathVariable("id") long id) {
52: 		itemRepository.deleteById(id);
53: 		return new ModelAndView("success");
54: 	}
55: }
