File: uaa/zoneadmin.sh

 2: set -x
 3: uaac target $UAA_ENDPOINT --skip-ssl-validation
 4: uaac token client get $ADMIN_CLIENT_ID -s $ADMIN_CLIENT_SECRET
 5: uaac client add temp --authorities zones.write,scim.zones --scope zones.$IDENTITY_ZONE_ID.admin --authorized_grant_types client_credentials,password -s $ZONEADMIN_CLIENT_SECRET
 6: uaac user add temp --email cf-spring-cloud-services@pivotal.io -p $ZONEADMIN_CLIENT_SECRET
 7: uaac member add zones.$IDENTITY_ZONE_ID.admin temp
 8: uaac token owner get temp temp -p $ZONEADMIN_CLIENT_SECRET -s $ZONEADMIN_CLIENT_SECRET
 9: uaac curl /oauth/clients -k -H "Content-type:application/json" -H "X-Identity-Zone-Id:$IDENTITY_ZONE_ID" -X POST -d "{\"client_id\":\"$ZONEADMIN_CLIENT_ID\",\"client_secret\":\"$ZONEADMIN_CLIENT_SECRET\",\"scope\":[\"uaa.none\"],\"resource_ids\":[\"none\"],\"authorities\":[\"uaa.admin\",\"clients.read\",\"clients.write\",\"scim.read\",\"scim.write\",\"clients.secret\"],\"authorized_grant_types\":[\"client_credentials\"]}"
10: uaac token client get $ADMIN_CLIENT_ID -s $ADMIN_CLIENT_SECRET
11: uaac user delete temp
12: uaac client delete temp
