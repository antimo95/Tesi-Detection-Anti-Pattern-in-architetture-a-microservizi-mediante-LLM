15:     private static final Logger LOG = LoggerFactory.getLogger(ProductAggregate.class);
16:     @AggregateIdentifier
17:     private String id;
18:     private String name;
19:     private boolean isSaleable = false;
20:     public ProductAggregate() {
21:     }
22:     @CommandHandler
23:     public ProductAggregate(AddProductCommand command) {
24:         LOG.debug("Command: 'AddProductCommand' received.");
25:         LOG.debug("Queuing up a new ProductAddedEvent for product '{}'", command.getId());
26:         apply(new ProductAddedEvent(command.getId(), command.getName()));
27:     }
28:     @CommandHandler
29:     public void markSaleable(MarkProductAsSaleableCommand command) {
30:         LOG.debug("Command: 'MarkProductAsSaleableCommand' received.");
31:         if (!this.isSaleable()) {
32:             apply(new ProductSaleableEvent(id));
33:         } else {
34:             throw new IllegalStateException("This ProductAggregate (" + this.getId() + ") is already Saleable.");
35:         }
36:     }
37:     @CommandHandler
38:     public void markUnsaleable(MarkProductAsUnsaleableCommand command) {
39:         LOG.debug("Command: 'MarkProductAsUnsaleableCommand' received.");
40:         if (this.isSaleable()) {
41:             apply(new ProductUnsaleableEvent(id));
42:         } else {
43:             throw new IllegalStateException("This ProductAggregate (" + this.getId() + ") is already off-sale.");
44:         }
45:     }
46:     @EventSourcingHandler
47:     public void on(ProductAddedEvent event) {
48:         this.id = event.getId();
49:         this.name = event.getName();
50:         LOG.debug("Applied: 'ProductAddedEvent' [{}] '{}'", event.getId(), event.getName());
51:     }

59:         this.isSaleable = false;
60:         LOG.debug("Applied: 'ProductUnsaleableEvent' [{}]", event.getId());
61:     }

64:     }

67:     }
