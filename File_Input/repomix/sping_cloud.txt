 1: package cloud.api.gateway.security;
 2: import org.springframework.security.oauth2.client.filter.OAuth2ClientContextFilter;
 3: import org.springframework.security.oauth2.client.resource.UserRedirectRequiredException;
 4: import org.springframework.security.web.DefaultRedirectStrategy;
 5: import org.springframework.security.web.RedirectStrategy;
 6: import org.springframework.web.util.UriComponentsBuilder;
 7: import javax.servlet.http.HttpServletRequest;
 8: import javax.servlet.http.HttpServletResponse;
 9: import java.io.IOException;
10: import java.util.Map;
11: class DynamicOauth2ClientContextFilter extends OAuth2ClientContextFilter {
12:     private RedirectStrategy redirectStrategy = new DefaultRedirectStrategy();
13:     @Override
14:     protected void redirectUser(UserRedirectRequiredException e, HttpServletRequest request,
15:                                 HttpServletResponse response) throws IOException {
16:         String redirectUri = e.getRedirectUri();
17:         UriComponentsBuilder builder = UriComponentsBuilder.fromUriString(redirectUri);
18:         Map<String, String> requestParams = e.getRequestParams();
19:         for (Map.Entry<String, String> param : requestParams.entrySet()) {
20:             builder.queryParam(param.getKey(), param.getValue());
21:         }
22:         if (e.getStateKey() != null) {
23:             builder.queryParam("state", e.getStateKey());
24:         }
25:         String url = getBaseUrl(request) + builder.build().encode().toUriString();
26:         this.redirectStrategy.sendRedirect(request, response, url);
27:     }
28:     @Override
29:     public void setRedirectStrategy(RedirectStrategy redirectStrategy) {
30:         this.redirectStrategy = redirectStrategy;
31:     }
32:     private String getBaseUrl(HttpServletRequest request) {
33:         StringBuffer url = request.getRequestURL();
34:         return  url.substring(0, url.length() - request.getRequestURI().length() + request.getContextPath().length());
35:     }
36: }

30:     private static final String CSRF_HEADER_NAME = "X-XSRF-TOKEN";
31:     @Bean
32:     @Primary
33:     public OAuth2ClientContextFilter dynamicOauth2ClientContextFilter() {
34:         return new DynamicOauth2ClientContextFilter();

36:     @Override

38:         http.authorizeRequests().antMatchers("/uaa/**", "/login").permitAll().anyRequest().authenticated()
39:             .and()

48:             private final Pattern allowedMethods = Pattern.compile("^(GET|HEAD|OPTIONS|TRACE)$");
49:             private final AntPathRequestMatcher[] requestMatchers = { new AntPathRequestMatcher("/uaa/**") };
50:             @Override
51:             public boolean matches(HttpServletRequest request) {
52:                 if (allowedMethods.matcher(request.getMethod()).matches()) {
53:                     return false;
54:                 }
55:                 for (AntPathRequestMatcher matcher : requestMatchers) {
56:                     if (matcher.matches(request)) {
57:                         return false;
58:                     }
59:                 }
60:                 return true;
61:             }
62:         };
63:     }
64:     private static Filter csrfHeaderFilter() {

70:                 if (csrf != null) {
71:                     Cookie cookie = new Cookie(CSRF_COOKIE_NAME, csrf.getToken());
72:                     cookie.setPath("/");
73:                     cookie.setSecure(true);
74:                     response.addCookie(cookie);
75:                 }
76:                 filterChain.doFilter(request, response);
77:             }
78:         };
79:     }

84:     }

 8: import org.springframework.beans.factory.annotation.Value;
 9: import org.springframework.context.annotation.Configuration;
10: import javax.annotation.PostConstruct;
11: import java.util.ArrayList;
12: import java.util.HashMap;
13: import java.util.List;
14: import java.util.Map;
15: import java.util.concurrent.ExecutorService;
16: import java.util.concurrent.Executors;
17: @Configuration
18: public class ZooKeeperConfig {
19:     @Value("${service.name}")
20:     String serviceName;
21:     @Value("${zookeeper.server.list}")
22:     String zookeeperList;
23:     ExecutorService executor = Executors.newSingleThreadExecutor();
24:     public static Map<String, UserService.Client> serviceMap = new HashMap<String, UserService.Client>();
25:     @PostConstruct
26:     private void init() {
27:         executor.execute(new Runnable() {
28:             @Override
29:             public void run() {
30:                 startZooKeeper();
31:                 try {
32:                     Thread.sleep(1000 * 60 * 60 * 24 * 360 * 10);
33:                 } catch (InterruptedException e) {
34:                     e.printStackTrace();
35:                 }
36:             }
37:         });
38:     }
39:     private void startZooKeeper() {
40:         List<String> currChilds = new ArrayList<String>();
41:         String servicePath = "/" + serviceName;
42:         ZkClient zkClient = new ZkClient(zookeeperList);
43:         boolean serviceExists = zkClient.exists(servicePath);
44:         if (serviceExists) {
45:             currChilds = zkClient.getChildren(servicePath);
46:         } else {
47:             throw new RuntimeException("service not exist!");
48:         }
49:         for (String instanceName : currChilds) {
50:             if (!serviceMap.containsKey(instanceName)) {
51:                 serviceMap.put(instanceName, createUserService(instanceName));
52:             }
53:         }
54:         zkClient.subscribeChildChanges(servicePath, new IZkChildListener() {
55:             public void handleChildChange(String parentPath,
56:                                           List<String> currentChilds) throws Exception {
57:                 for (String instanceName : currentChilds) {
58:                     if (!serviceMap.containsKey(instanceName)) {
59:                         serviceMap.put(instanceName,createUserService(instanceName));
60:                     }
61:                 }
62:                 for (Map.Entry<String, UserService.Client> entry : serviceMap.entrySet()) {
63:                     if (!currentChilds.contains(entry.getKey())) {
64:                         UserService.Client c = serviceMap.get(entry.getKey());
65:                         try {
66:                             c.getInputProtocol().getTransport().close();
67:                             c.getOutputProtocol().getTransport().close();
68:                         } catch (Exception e) {
69:                             e.printStackTrace();
70:                         }
71:                         serviceMap.remove(entry.getKey());
72:                     }
73:                 }
74:                 System.out.println(parentPath + "事件触发");
75:             }
76:         });
77:     }
78:     private UserService.Client createUserService(String serviceInstanceName) {
79:         String ip = serviceInstanceName.split("-")[1];
80:         TSocket transport = new TSocket(ip, 7911);
81:         try {
82:             transport.open();
83:         } catch (TTransportException e) {
84:             e.printStackTrace();
85:         }
86:         return new UserService.Client(new TBinaryProtocol(transport));
87:     }
88: }
